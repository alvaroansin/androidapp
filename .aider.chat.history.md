package com.example.ytplayer

import okhttp3.OkHttpClient
import okhttp3.Request
import org.json.JSONObject
import java.util.concurrent.TimeUnit

object PipedApi {
    // Puedes cambiar la instancia si esta no funciona:
    // https://github.com/TeamPiped/Piped/wiki/Instances
    private const val BASE = "https://pipedapi.kavin.rocks"

    private val client = OkHttpClient.Builder()
        .connectTimeout(15, TimeUnit.SECONDS)
        .readTimeout(15, TimeUnit.SECONDS)
        .build()

    private fun get(url: String): String {
        val request = Request.Builder()
            .url(url)
            .header("User-Agent", "Mozilla/5.0")
            .build()
        client.newCall(request).execute().use { response ->
            if (!response.isSuccessful) throw Exception("HTTP ${response.code}")
            return response.body!!.string()
        }
    }

    fun search(query: String): List<Video> {
        val json = JSONObject(get("$BASE/search?q=${java.net.URLEncoder.encode(query, "UTF-8")}&filter=videos"))
        val items = json.getJSONArray("items")
        val result = mutableListOf<Video>()
        for (i in 0 until items.length()) {
            val item = items.getJSONObject(i)
            val url = item.optString("url", "")
            val id = url.substringAfter("v=", "")
            if (id.isBlank()) continue
            result.add(
                Video(
                    id = id,
                    title = item.optString("title", ""),
                    uploader = item.optString("uploaderName", ""),
                    duration = item.optInt("duration", 0),
                    thumbnail = item.optString("thumbnail", "")
                )
            )
        }
        return result
    }

    fun getStreamUrl(videoId: String): String {
        val json = JSONObject(get("$BASE/streams/$videoId"))
        val videoStreams = json.getJSONArray("videoStreams")
        var bestUrl: String? = null
        var bestQuality = -1
        for (i in 0 until videoStreams.length()) {
            val stream = videoStreams.getJSONObject(i)
            val videoOnly = stream.optBoolean("videoOnly", true)
            if (videoOnly) continue // Solo queremos streams combinados (audio + video)
            val quality = stream.optString("quality", "0p").removeSuffix("p").toIntOrNull() ?: 0
            if (quality > bestQuality) {
                bestQuality = quality
                bestUrl = stream.optString("url", "")
            }
        }
        if (bestUrl == null) throw Exception("No se encontró stream reproducible")
        return bestUrl
    }
}

package com.example.ytplayer

import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.widget.ImageView
import android.widget.TextView
import androidx.recyclerview.widget.RecyclerView
import coil.load

class VideoAdapter(
    private val videos: List<Video>,
    private val onClick: (Video) -> Unit
) : RecyclerView.Adapter<VideoAdapter.ViewHolder>() {

    class ViewHolder(view: View) : RecyclerView.ViewHolder(view) {
        val thumbnail: ImageView = view.findViewById(R.id.thumbnail)
        val title: TextView = view.findViewById(R.id.title)
        val uploader: TextView = view.findViewById(R.id.uploader)
        val duration: TextView = view.findViewById(R.id.duration)
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): ViewHolder {
        val view = LayoutInflater.from(parent.context)
            .inflate(R.layout.item_video, parent, false)
        return ViewHolder(view)
    }

    override fun onBindViewHolder(holder: ViewHolder, position: Int) {
        val video = videos[position]
        holder.title.text = video.title
        holder.uploader.text = video.uploader
        holder.duration.text = if (video.duration > 0) {
            "%d:%02d".format(video.duration / 60, video.duration % 60)
        } else ""
        holder.thumbnail.load(video.thumbnail) {
            crossfade(true)
        }
        holder.itemView.setOnClickListener { onClick(video) }
    }

    override fun getItemCount() = videos.size
}

package com.example.ytplayer

import android.content.Intent
import android.os.Bundle
import android.widget.Button

import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import kotlin.concurrent.thread

class MainActivity : AppCompatActivity() {

    private val videos = mutableListOf<Video>()
    private lateinit var adapter: VideoAdapter

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        val searchInput = findViewById<EditText>(R.id.searchInput)
        val searchButton = findViewById<Button>(R.id.searchButton)
        val recyclerView = findViewById<RecyclerView>(R.id.recyclerView)

        adapter = VideoAdapter(videos) { video ->
            val intent = Intent(this, PlayerActivity::class.java)
            intent.putExtra("videoId", video.id)
            intent.putExtra("title", video.title)
            startActivity(intent)
        }
        recyclerView.layoutManager = LinearLayoutManager(this)
        recyclerView.adapter = adapter

        searchButton.setOnClickListener {
            val query = searchInput.text.toString().trim()
            if (query.isNotEmpty()) {
                searchVideos(query)
            }
        }
    }

    private fun searchVideos(query: String) {
        Toast.makeText(this, getString(R.string.searching), Toast.LENGTH_SHORT).show()
        thread {
            try {
                val results = PipedApi.search(query)
                runOnUiThread {
                    videos.clear()
                    videos.addAll(results)
                    adapter.notifyDataSetChanged()
                }
            } catch (e: Exception) {
                runOnUiThread {
                    Toast.makeText(this, getString(R.string.search_error, e.message), Toast.LENGTH_LONG).show()
                }
            }
        }
    }
}

package com.example.ytplayer

import androidx.media3.common.AudioAttributes
import androidx.media3.common.C
import androidx.media3.exoplayer.ExoPlayer
import androidx.media3.session.MediaSession
import androidx.media3.session.MediaSessionService

/**
 * Servicio en primer plano que mantiene la reproducción activa
 * incluso con la pantalla apagada o la app en segundo plano.
 */
class PlayerService : MediaSessionService() {

    private var mediaSession: MediaSession? = null

    override fun onCreate() {
        super.onCreate()
        val player = ExoPlayer.Builder(this)
            .setAudioAttributes(
                AudioAttributes.Builder()
                    .setContentType(C.AUDIO_CONTENT_TYPE_MOVIE)
                    .setUsage(C.USAGE_MEDIA)
                    .build(),
                true
            )
            .setHandleAudioBecomingNoisy(true)
            .setWakeMode(C.WAKE_MODE_NETWORK)
            .build()
        mediaSession = MediaSession.Builder(this, player).build()
    }

    override fun onGetSession(controllerInfo: MediaSession.ControllerInfo): MediaSession? {
        return mediaSession
    }

    override fun onDestroy() {
        mediaSession?.run {
            player.release()
            release()
        }
        mediaSession = null
        super.onDestroy()
    }
}

package com.example.ytplayer

import android.os.Bundle
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import androidx.media3.common.MediaItem
import androidx.media3.common.MediaMetadata
import androidx.media3.session.MediaController
import androidx.media3.session.SessionToken
import androidx.media3.ui.PlayerView
import kotlin.concurrent.thread

class PlayerActivity : AppCompatActivity() {

    private var mediaController: MediaController? = null
    private lateinit var playerView: PlayerView

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_player)

        playerView = findViewById(R.id.playerView)

        val videoId = intent.getStringExtra("videoId") ?: return
        val title = intent.getStringExtra("title") ?: ""

        // Conectar con el servicio (el audio sigue con la pantalla apagada)
        val sessionToken = SessionToken(this, android.content.ComponentName(this, PlayerService::class.java))
        val controllerFuture = MediaController.Builder(this, sessionToken).buildAsync()
        controllerFuture.addListener({
            try {
                mediaController = controllerFuture.get()
                playerView.player = mediaController
                loadAndPlay(videoId, title)
            } catch (e: Exception) {
                Toast.makeText(this, getString(R.string.player_error, e.message), Toast.LENGTH_LONG).show()
            }
        }, androidx.media3.common.util.UnstableApi::class.java.let { Runnable { it } }.let { mainExecutor }) { }
    }

    private fun loadAndPlay(videoId: String, title: String) {
        thread {
            try {
                val streamUrl = PipedApi.getStreamUrl(videoId)
                runOnUiThread {
                    val mediaItem = MediaItem.Builder()
                        .setUri(streamUrl)
                        .setMediaId(videoId)
                        .setMediaMetadata(
                            MediaMetadata.Builder()
                                .setTitle(title)
                                .build()
                        )
                        .build()
                    mediaController?.let {
                        it.setMediaItem(mediaItem)
                        it.prepare()
                        it.play()
                    }
                }
            } catch (e: Exception) {
                runOnUiThread {
                    Toast.makeText(this, getString(R.string.stream_error, e.message), Toast.LENGTH_LONG).show()
                }
            }
        }
    }

    override fun onDestroy() {
        // Solo liberamos el controlador; el servicio sigue reproduciendo en segundo plano
        mediaController?.release()
        mediaController = null
        playerView.player = null
        super.onDestroy()
    }
}

1. **MainActivity** busca vídeos mediante la API de Piped y muestra los resultados en un RecyclerView con miniaturas (Coil).
2. **PlayerActivity** se conecta al servicio mediante un `MediaController` de Media3, obtiene la URL del stream (sin anuncios, ya que Piped sirve los streams directamente) y lo reproduce.
3. **PlayerService** es un `MediaSessionService` en primer plano: mantiene el ExoPlayer vivo con la pantalla apagada, muestra la notificación con controles multimedia y respeta el modo "no molestar" del audio (pausa al desconectar auriculares).
